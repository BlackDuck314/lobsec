```python
class RetryConfig(BaseModel):
    max_attempts: int = Field(ge=1, le=10, default=3)
    initial_delay_ms: int = Field(ge=100, default=2000)
    max_delay_ms: int = Field(ge=1000, default=30000)
    backoff_multiplier: float = Field(ge=1.0, le=5.0, default=2.0)
    jitter: bool = True
    skip_on_403: bool = False

class Mission(BaseModel):
    schema_version: str = "1.0"
    name: str
    description: str
    type: Literal["http_download", "browser_scrape", "api_call"]
    frequency: Literal["daily", "weekly", "monthly", "quarterly"]
    priority: int = Field(ge=1, le=5)
    source: dict
    extraction: dict
    output: dict
    retry: RetryConfig = Field(default_factory=RetryConfig)
    timeout_ms: int = Field(ge=5000, le=3600000, default=120000)
    concurrency: ConcurrencyConfig | None = None
    areas: list[str] | None = None

def load_mission(path: str) -> Mission
def load_all_missions(directory: str) -> dict[str, Mission]
```
