       ```typescript
       export interface NormalizationInput {
         filePath: string;
         source: string;

       export interface NormalizationResult {
         source: string;
         recordCount: number;
         measurementDateRange: { start: string; end: string };
         gapWarnings: GapWarning[];
         volumeWarning?: VolumeWarning;
       }

       export interface GapWarning {
         source: string;
         lastDate: string;
         expectedFrequency: CollectionFrequency;
         gapDays: number;
         isStale: boolean;  // true if gap > 2x expected
       }

       export interface VolumeWarning {
         source: string;
         currentCount: number;
         rollingAverage: number;
         ratio: number;  // current / average
       }

       // Map source name to Python normalization module
       export const SOURCE_MODULE_MAP: Record<string, string> = {
         "dld-sales": "normalize_dld",
         "ejari-rentals": "normalize_ejari",
         "building-permits": "normalize_permits",
         "adrec-abu-dhabi": "normalize_adrec",
         "bayut-listings": "normalize_bayut",
         "propertyfinder-listings": "normalize_propertyfinder",
         "dewa-connections": "normalize_dewa",
       };
       ```

       ```typescript
       export type PythonScriptName =

         | "normalize_adrec"
         | "normalize_bayut"

         | "normalize_dewa";
       ```

    3. **Create normalization/orchestrator.ts** — `normalizeCollectionResult()`:
       - Input: `(db, source, collectionResult: CollectionResult)` — called after each successful collection
       - Look up Python module in SOURCE_MODULE_MAP. Throw if source not found.
       - Call `runPython()` with `{filePath, source, collectedAt}` input (per bridge pattern)
       - Python module returns `NormalizedRecord[]` where each record has: `{source, measurement_date, metric_name, value, available_date}`
       - **Upsert semantics** (per user decision): Within a db.transaction():
         1. Compute min/max measurement_date from returned records
         2. DELETE FROM normalized_monthly WHERE source=? AND measurement_date BETWEEN ? AND ?
         3. INSERT all records using insertNormalized()
       - Add `deleteNormalizedRange(db, source, startDate, endDate)` to queries.ts for the DELETE step
       - After upsert, run gap detection and volume validation
       - Return NormalizationResult with gap and volume warnings
       - **Timeout**: 120s for normalization Python calls (larger datasets)

    4. **Create normalization/gap-detection.ts** — `detectGaps()`:
       - Input: `(db, source, expectedFrequency: CollectionFrequency)`
       - Query collection_log for the last N successful runs for this source
       - Compute expected interval: daily=1, weekly=7, monthly=30, quarterly=90 (days)
       - If latest successful run is more than 2x expected interval ago, flag as STALE
       - Return GapWarning[] (may be empty if no gaps)

    5. **Create normalization/volume-validation.ts** — `validateVolume()`:
       - Input: `(db, source, currentRowCount: number)`
       - Query collection_log for the last 4 successful collections (per user decision: N=4 baseline)
       - If fewer than 4 successful runs exist, skip validation (return null — baseline not yet established)
       - Compute rolling average of row_count from those 4 runs
       - If currentRowCount < 50% of rolling average, return VolumeWarning
       - Otherwise return null

    6. **Export everything from index.ts** — Add exports for:
       - `normalizeCollectionResult` from normalization/orchestrator
       - `detectGaps` from normalization/gap-detection
       - `validateVolume` from normalization/volume-validation
       - All types from normalization/types
       - `initAreaTable`, `getCanonicalArea` from areas/mapping
