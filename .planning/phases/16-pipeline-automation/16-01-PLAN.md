```python
def adapt_scraper_format(raw_data):
    """Convert Ninja Scraper list-of-pages format to normalizer dict format."""
    if isinstance(raw_data, dict):
        return raw_data  # Already in expected format

    if not isinstance(raw_data, list):
        raise ValueError(f"Unexpected data type: {type(raw_data).__name__}")

    # Merge cards from all pages into single listings list
    all_cards = []
    total_count = 0
    for page in raw_data:
        cards = page.get("cards") or []
        all_cards.extend(cards)
        # total_count may be string like "1,000+" or int
        tc = page.get("total_count")
        if tc is not None:
            if isinstance(tc, str):
                # Parse "1,000+" -> 1000
                tc = int(tc.replace(",", "").replace("+", "").strip() or "0")
            total_count = max(total_count, tc)

    # If total_count not available, use card count
    if total_count == 0:
        total_count = len(all_cards)

    return {"listings": all_cards, "total_count": total_count}
```

```python
def adapt_scraper_format(raw_data: list | dict, collected_at: str) -> dict:
    """Convert Ninja Scraper area-list format to normalizer dict format."""
    if isinstance(raw_data, dict) and "scrapedAt" in raw_data:
        return raw_data  # Already in expected format

    if not isinstance(raw_data, list):
        raise ValueError(f"Unexpected data type: {type(raw_data).__name__}")

    areas = []
    for item in raw_data:
        cards = item.get("cards") or []

        # Convert card fields to normalizer format
        listings = []
        for card in cards:
            # Parse price string "33,500,000 AED" -> float
            price_str = card.get("price", "")
            price = None
            if price_str and isinstance(price_str, str):
                try:
                    price = float(price_str.replace(",", "").replace("AED", "").strip())
                except ValueError:
                    price = None
            elif isinstance(price_str, (int, float)):
                price = float(price_str)

            # Parse bedrooms string "4" -> int
            bedrooms = None
            bed_val = card.get("bedrooms")
            if bed_val is not None:
                try:
                    bedrooms = int(str(bed_val).strip())

                    bedrooms = None

            # Parse sqft string "4,567 sqft" -> float
            sqft = None
            sqft_str = card.get("area_sqft", "")
            if sqft_str and isinstance(sqft_str, str):
                try:
                    sqft = float(sqft_str.replace(",", "").replace("sqft", "").strip())
                except ValueError:
                    sqft = None
            elif isinstance(sqft_str, (int, float)):
                sqft = float(sqft_str)

            listings.append({
                "price": price,
                "bedrooms": bedrooms,
                "sqft": sqft,
                "type": card.get("property_type"),
                "reducedPrice": False,  # PropertyFinder cards don't have reduced badge in current scraper
```

```python
        listings = []
        for card in cards:
            # Parse price (may be string "1,500,000 AED" or None)
            price = None
            price_str = card.get("price", "")
            if price_str and isinstance(price_str, str):
                try:
                    price = float(price_str.replace(",", "").replace("AED", "").strip())
                except ValueError:
                    price = None
            elif isinstance(price_str, (int, float)):
                price = float(price_str)

            # Parse bedrooms
            bedrooms = None
            bed_val = card.get("bedrooms")
            if bed_val is not None:
                try:
                    bedrooms = int(str(bed_val).strip())
                except ValueError:

            # Parse sqft (Bayut uses "sqft" not "area_sqft")
            sqft = None
            sqft_str = card.get("sqft") or card.get("area_sqft", "")
            if sqft_str and isinstance(sqft_str, str):
                try:
                    sqft = float(sqft_str.replace(",", "").replace("sqft", "").strip())
                except ValueError:
                    sqft = None
            elif isinstance(sqft_str, (int, float)):
                sqft = float(sqft_str)

            listings.append({
                "price": price,
                "bedrooms": bedrooms,
                "sqft": sqft,
                "type": card.get("property_type"),
                "reducedPrice": bool(card.get("reduced_badge") or card.get("reducedPrice", False)),
            })

        areas.append({
            "area": item.get("area", "unknown"),
            "listingCount": item.get("listing_count") or item.get("card_count") or len(cards),

        "scrapedAt": collected_at,
```
